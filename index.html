// ============================================================ // COLE TODO ESTE CÓDIGO NO APPS SCRIPT DA SUA PLANILHA // Extensões → Apps Script → apague tudo → cole → salve // Depois: Implantar → Nova implantação → Web App // Executar como: Eu | Quem tem acesso: Qualquer pessoa // ============================================================ const SHEET_AGENDAMENTOS = "Agendamentos"; const SHEET_HORARIOS = "Horarios"; function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { const params = e.parameter || {}; const action = params.action; let result; try { if (action === "getSlots") { result = getSlots(params.date); } else if (action === "getBookings") { result = getBookings(); } else if (action === "addBooking") { result = addBooking(params); } else if (action === "cancelBooking") { result = cancelBooking(params.id); } else if (action === "addSlot") { result = addSlot(params.date, params.time); } else if (action === "removeSlot") { result = removeSlot(params.date, params.time); } else { result = { ok: false, error: "Ação desconhecida: " + action }; } } catch (err) { result = { ok: false, error: err.toString() }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ── Inicializa abas se não existirem ────────────────────── function ensureSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let shAg = ss.getSheetByName(SHEET_AGENDAMENTOS); if (!shAg) { shAg = ss.insertSheet(SHEET_AGENDAMENTOS); shAg.appendRow(["ID", "Nome", "Telefone", "Servico", "Data", "Horario", "Obs", "CriadoEm"]); shAg.setFrozenRows(1); } let shHo = ss.getSheetByName(SHEET_HORARIOS); if (!shHo) { shHo = ss.insertSheet(SHEET_HORARIOS); shHo.appendRow(["Data", "Horario"]); shHo.setFrozenRows(1); } return { shAg, shHo }; } // ── Buscar horários disponíveis por data ────────────────── function getSlots(date) { const { shAg, shHo } = ensureSheets(); // Todos os horários cadastrados para a data const hoData = shHo.getDataRange().getValues(); const allSlots = hoData.slice(1) .filter(r => String(r[0]).trim() === String(date).trim()) .map(r => String(r[1]).trim()) .filter(Boolean); // Horários já reservados na data const agData = shAg.getDataRange().getValues(); const booked = agData.slice(1) .filter(r => String(r[4]).trim() === String(date).trim()) .map(r => String(r[5]).trim()); const slots = allSlots.map(t => ({ time: t, available: !booked.includes(t) })); return { ok: true, slots }; } // ── Buscar todos os agendamentos ────────────────────────── function getBookings() { const { shAg } = ensureSheets(); const rows = shAg.getDataRange().getValues(); const bookings = rows.slice(1) .filter(r => r[0]) .map(r => ({ id: String(r[0]), name: r[1], phone: r[2], service: r[3], date: r[4], time: r[5], obs: r[6], createdAt: r[7] })) .sort((a, b) => (a.date + a.time) > (b.date + b.time) ? 1 : -1); return { ok: true, bookings }; } // ── Adicionar agendamento ───────────────────────────────── function addBooking(p) { const { shAg } = ensureSheets(); // Verificar conflito const rows = shAg.getDataRange().getValues(); const conflict = rows.slice(1).find(r => String(r[4]).trim() === String(p.date).trim() && String(r[5]).trim() === String(p.time).trim() ); if (conflict) { return { ok: false, error: "Horário já reservado" }; } const id = Date.now().toString(); shAg.appendRow([id, p.name, p.phone, p.service, p.date, p.time, p.obs || "", new Date().toISOString()]); return { ok: true, id }; } // ── Cancelar agendamento ────────────────────────────────── function cancelBooking(id) { const { shAg } = ensureSheets(); const rows = shAg.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(id)) { shAg.deleteRow(i + 1); return { ok: true }; } } return { ok: false, error: "Agendamento não encontrado" }; } // ── Adicionar horário disponível ────────────────────────── function addSlot(date, time) { const { shHo } = ensureSheets(); const rows = shHo.getDataRange().getValues(); const exists = rows.slice(1).find(r => String(r[0]).trim() === String(date).trim() && String(r[1]).trim() === String(time).trim() ); if (exists) return { ok: false, error: "Horário já existe" }; shHo.appendRow([date, time]); return { ok: true }; } // ── Remover horário disponível ──────────────────────────── function removeSlot(date, time) { const { shHo } = ensureSheets(); const rows = shHo.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === String(date).trim() && String(rows[i][1]).trim() === String(time).trim()) { shHo.deleteRow(i + 1); return { ok: true }; } } return { ok: false, error: "Horário não encontrado" }; }
